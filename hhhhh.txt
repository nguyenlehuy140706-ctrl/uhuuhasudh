    <script>
        "use strict";

        const searchInput = document.getElementById("searchInput");
        const courseCards = Array.from(
            document.querySelectorAll(".course-card")
        );
        const categoryButtons = Array.from(
            document.querySelectorAll(".category-btn")
        );
        const emptyMessage = document.getElementById("emptyMessage");

        let selectedCategory = "all";

        function applyFilters() {
            const keyword = searchInput.value
                .trim()
                .toLocaleLowerCase("vi");

            let visibleCount = 0;

            courseCards.forEach(function (card) {
                const title = card.querySelector("h3").textContent
                    .toLocaleLowerCase("vi");

                const instructor = card.querySelector(".instructor")
                    .textContent.toLocaleLowerCase("vi");

                const category = card.dataset.category;

                const matchesKeyword =
                    title.includes(keyword) ||
                    instructor.includes(keyword);

                const matchesCategory =
                    selectedCategory === "all" ||
                    category === selectedCategory;

                const visible = matchesKeyword && matchesCategory;

                card.hidden = !visible;

                if (visible) {
                    visibleCount++;
                }
            });

            emptyMessage.style.display =
                visibleCount === 0 ? "block" : "none";
        }

        searchInput.addEventListener("input", applyFilters);

        categoryButtons.forEach(function (button) {
            button.addEventListener("click", function () {
                selectedCategory = button.dataset.category;

                categoryButtons.forEach(function (item) {
                    const active =
                        item.dataset.category === selectedCategory;

                    item.classList.toggle("active", active);
                    item.setAttribute("aria-pressed", String(active));
                });

                applyFilters();
            });
        });

        document.getElementById("exploreBtn")
            .addEventListener("click", function () {
                document.getElementById("courses").scrollIntoView({
                    behavior: "smooth"
                });
            });

        document.getElementById("loginBtn")
            .addEventListener("click", function () {
                alert("Đây là giao diện đăng nhập minh họa.");
            });

        document.getElementById("signupBtn")
            .addEventListener("click", function () {
                alert("Đây là giao diện đăng ký tài khoản minh họa.");
            });

        document.querySelectorAll(".enroll-btn")
            .forEach(function (button) {
                button.addEventListener("click", function () {
                    const courseName = button.dataset.course;

                    alert(
                        "Bạn đã chọn khóa học: " + courseName +
                        "\nĐây là bản demo, chưa có chức năng thanh toán thực tế."
                    );
                });
            });

        // Hiển thị toàn bộ khóa học khi trang được mở.
        applyFilters();
    </script>
